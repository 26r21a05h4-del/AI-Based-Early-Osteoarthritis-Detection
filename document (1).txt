// 1. Vector Goniometer Math
export function calculateJointAngle(pointA, pointB, pointC) {
  if (!pointA || !pointB || !pointC) return 180;
  const vBAx = pointA.x - pointB.x;
  const vBAy = pointA.y - pointB.y;
  const vBCx = pointC.x - pointB.x;
  const vBCy = pointC.y - pointB.y;
  const dot = vBAx * vBCx + vBAy * vBCy;
  const mag = Math.sqrt(vBAx * vBAx + vBAy * vBAy) * Math.sqrt(vBCx * vBCx + vBCy * vBCy);
  if (mag === 0) return 180;
  return Math.round((Math.acos(Math.max(-1, Math.min(1, dot / mag))) * 180) / Math.PI);
}

// 2. Clinical Scoring & Risk Stratification
export function calculateScreeningScore(symptoms, metrics) {
  let raw = 0;
  if (symptoms.kneePain === 'mild') raw += 8;
  else if (symptoms.kneePain === 'moderate') raw += 18;
  else if (symptoms.kneePain === 'severe') raw += 28;
  const score = Math.round(Math.min(100, (raw / 90) * 40 + (100 - metrics.postureScore) * 0.18 + (100 - metrics.walkingSymmetryScore) * 0.22 + (100 - metrics.kneeMovementScore) * 0.2));
  return { totalScore: score, riskLevel: score >= 66 ? 'HIGH RISK' : score >= 33 ? 'MEDIUM RISK' : 'LOW RISK' };
}

// 3. Permanent Audit Trail Storage (Deletion permanently disabled)
const KEY = 'oa_screening_history_v1';
export function getStoredScreenings() {
  try { return JSON.parse(localStorage.getItem(KEY) || '[]'); } catch { return []; }
}
export function saveScreeningResult(res) {
  const all = [res, ...getStoredScreenings().filter(i => i.id !== res.id)];
  localStorage.setItem(KEY, JSON.stringify(all));
}
